% 출처: L01 (MLP, XOR) — p32-34, p36
\part[3] What is the minimum number of layers (counting hidden and output layers only) required for a neural network to solve the XOR problem?

\begin{oneparchoices}
 \choice 1
 \choice 2
 \choice 3
 \choice 4
\end{oneparchoices}

\answer{B. XOR requires at least 1 hidden layer + 1 output layer = 2 layers. \textit{(L01 p32--34, p36)}}

% 출처: L02 (Cross Entropy calculation) — p8-9
\part[4] A classifier outputs the softmax probability $[0.1,\; 0.2,\; 0.3,\; 0.4]$ for a 4-class classification problem. If the ground-truth class is class 3, what is the cross-entropy loss?

\begin{oneparchoices}
 \choice $-\log(0.1)$
 \choice $-\log(0.3)$
 \choice $-\log(0.4)$
 \choice $-\sum_{c=1}^{4} \log(p_c)$
\end{oneparchoices}

\answer{B. With one-hot ground truth, CE loss $= -\log(\hat{p}_{y_i})$. Since class 3 has $\hat{p}_3 = 0.3$, the loss is $-\log(0.3)$. \textit{(L02 p8--9)}}
\vspace{\stretch{1}}

% 출처: L03 (Softmax + CE gradient) — p72-73
\part[4] Given the softmax output $\mathbf{p} = [0.2,\; 0.5,\; 0.3]$ and the ground-truth class is class 2, what is the gradient of the cross-entropy loss with respect to the logit vector $\mathbf{s}$?

\begin{oneparchoices}
 \choice $[0.2,\; -0.5,\; 0.3]$
 \choice $[0.2,\; 0.5,\; -0.7]$
 \choice $[-0.8,\; 0.5,\; 0.3]$
 \choice $[0.2,\; 0.5,\; 0.3]$
\end{oneparchoices}

\answer{A. $\partial \mathcal{L} / \partial \mathbf{s} = \mathbf{p} - \mathbf{y} = [0.2,\; 0.5,\; 0.3] - [0,\; 1,\; 0] = [0.2,\; -0.5,\; 0.3]$. \textit{(L03 p72--73)}}
\vspace{\stretch{1}}

% 출처: L03 (Max gate) — p28
\part[3] In a computational graph, a max gate routes the upstream gradient to \rule{1cm}{0.15mm}.

\begin{oneparchoices}
 \choice both inputs equally
 \choice the larger input only
 \choice the smaller input only
 \choice neither input
\end{oneparchoices}

\answer{B. The max gate passes the gradient only to the input that was selected (the larger one) and assigns zero gradient to the other input. \textit{(L03 p28)}}
\vspace{\stretch{1}}

% 출처: L04 (Optimizer identification) — p23-24
\part[4] Which optimizer uses an \textbf{exponential moving average of squared gradients} to adapt the learning rate for each parameter?

\begin{oneparchoices}
 \choice SGD with Momentum
 \choice AdaGrad
 \choice RMSProp
 \choice Nesterov Momentum
\end{oneparchoices}

\answer{C. RMSProp uses $v_t = \rho \, v_{t-1} + (1-\rho) \, g_t^2$ (exponential moving average). AdaGrad uses a cumulative sum, not a moving average. Momentum and Nesterov do not adapt per-parameter learning rates. \textit{(L04 p23--24)}}
\vspace{\stretch{1}}

% 출처: L04 (Mini-batch SGD) — p4-5
\part[3] What is the main advantage of using mini-batch SGD over full-batch gradient descent?

\begin{oneparchoices}
 \choice It always converges to the global minimum. \\
 \choice It is computationally faster per update by approximating the gradient. \\
 \choice It computes the exact gradient of the full dataset. \\
 \choice It eliminates the need for a learning rate. \\
\end{oneparchoices}

\answer{B. Mini-batch SGD approximates the gradient using a subset, making each update much faster than computing the gradient over the entire dataset. \textit{(L04 p4--5)}}
\vspace{\stretch{1}}

% 출처: L05 (Dead neuron) — p2, p9
\part[3] Which activation function can suffer from the ``dead neuron'' problem?

\begin{oneparchoices}
 \choice Sigmoid
 \choice Tanh
 \choice ReLU
 \choice Softmax
\end{oneparchoices}

\answer{C. ReLU outputs 0 for all negative inputs and its gradient is 0 in that region, so a neuron can become permanently dead. Sigmoid and Tanh have vanishing but never exactly zero gradients. \textit{(L05 p2, p9)}}
\vspace{\stretch{1}}

% 출처: L06 (F1 score calculation) — p4-5, p12
\part[4] A binary classifier produces the following results: $\text{TP} = 40$, $\text{FP} = 10$, $\text{FN} = 10$, $\text{TN} = 40$. What is the F1 score?

\begin{oneparchoices}
 \choice 0.75
 \choice 0.80
 \choice 0.85
 \choice 0.90
\end{oneparchoices}

\answer{B. Precision $= 40/50 = 0.8$, Recall $= 40/50 = 0.8$, $F_1 = 2 \times 0.8 \times 0.8 / (0.8 + 0.8) = 0.80$. \textit{(L06 p4--5, p12)}}
\vspace{\stretch{1}}

% 출처: L07 (Convolution output size) — p54-55
\part[4] Given an input of size $32 \times 32 \times 3$, what is the output size after a convolution layer with 16 filters of size $5 \times 5$, stride 1, and padding 0?

\begin{oneparchoices}
 \choice $28 \times 28 \times 3$
 \choice $28 \times 28 \times 16$
 \choice $32 \times 32 \times 16$
 \choice $28 \times 28 \times 5$
\end{oneparchoices}

\answer{B. Output spatial size $= (32 - 5)/1 + 1 = 28$. Output depth $=$ number of filters $= 16$. So $28 \times 28 \times 16$. \textit{(L07 p54--55)}}
\vspace{\stretch{1}}

% 출처: L07 (Same padding) — p55
\part[4] What padding size is required for a $3 \times 3$ filter to produce an output with the same spatial dimensions as the input? (stride = 1)

\begin{oneparchoices}
 \choice 0
 \choice 1
 \choice 2
 \choice 3
\end{oneparchoices}

\answer{B. Same padding: $p = (k-1)/2 = (3-1)/2 = 1$. \textit{(L07 p55)}}
\vspace{\stretch{1}}

% 출처: L01 (Forward propagation) — p39-44, p46
\part[3] In a feedforward neural network, the forward propagation computes the output by \rule{1cm}{0.15mm}.

\begin{oneparchoices}
 \choice propagating signals from the output layer to the input layer \\
 \choice propagating signals from the input layer to the output layer \\
 \choice updating the weights using the chain rule \\
 \choice computing the gradient of the loss function \\
\end{oneparchoices}

\answer{B. Forward propagation computes the output by passing signals layer-by-layer from the input to the output. \textit{(L01 p39--44, p46)}}
\vspace{\stretch{1}}

% 출처: L02 (Softmax) — p6
\part[3] Given logits $[2.0,\; 1.0,\; 0.1]$, which class will have the \textbf{highest} softmax probability?

\begin{oneparchoices}
 \choice Class 1
 \choice Class 2
 \choice Class 3
 \choice All classes have equal probability
\end{oneparchoices}

\answer{A. Softmax is a monotonic function—the largest logit always receives the highest probability. \textit{(L02 p6)}}
\vspace{\stretch{1}}

% 출처: L03 (Add gate) — p28
\part[3] An add gate in a computational graph works as a gradient \rule{1cm}{0.15mm}.

\begin{oneparchoices}
 \choice distributor
 \choice switcher
 \choice amplifier
 \choice None of the above
\end{oneparchoices}

\answer{A. An add gate distributes the upstream gradient equally to both inputs ($\partial(x+y)/\partial x = 1$). \textit{(L03 p28)}}

% 출처: L04 (Adam) — p25, p28-29
\part[4] Which of the following is \textbf{NOT} true about the Adam optimizer?

\begin{oneparchoices}
 \choice It uses a moving average to estimate the first moment (mean) of gradients. \\
 \choice It combines AdaGrad and Momentum. \\
 \choice It applies bias correction to the moment estimates. \\
 \choice None of the above. \\
\end{oneparchoices}

\answer{B. Adam combines \textbf{RMSProp} (not AdaGrad) and Momentum, with bias correction. \textit{(L04 p25, p28--29)}}
\vspace{\stretch{1}}

% 출처: L05 (Dropout) — p25-26
\part[3] During test time, a neural network with inverted dropout \rule{1cm}{0.15mm}.

\begin{oneparchoices}
 \choice randomly drops neurons as in training \\
 \choice uses all neurons without any modification \\
 \choice scales the weights by the dropout rate \\
 \choice removes the dropout layers entirely from the network \\
\end{oneparchoices}

\answer{B. With inverted dropout, weights are already scaled during training ($/ p$), so at test time all neurons are used without modification. \textit{(L05 p25--26)}}

% 출처: L05 (Sigmoid) — p3

 \choice $[-1, 1]$
 \choice $[0, 1]$
 \choice $(-\infty, \infty)$

% 출처: L06 (Precision / Recall) — p4-5
\part[3] Precision is defined as:

\begin{oneparchoices}
 \choice $\frac{\text{TP}}{\text{TP} + \text{FN}}$
 \choice $\frac{\text{TP}}{\text{TP} + \text{FP}}$
 \choice $\frac{\text{TP} + \text{TN}}{\text{Total}}$
 \choice $\frac{\text{TP}}{\text{Total}}$
\end{oneparchoices}

\answer{B. Precision = TP / (TP + FP). Option A is the definition of Recall. \textit{(L06 p4--5)}}
\vspace{\stretch{1}}

% 출처: L06 (Micro vs Macro averaging) — p42, p44
\part[4] In multi-class evaluation, micro-averaging computes the metric by:

\begin{oneparchoices}
 \choice computing the metric for each class, then averaging \\
 \choice aggregating all TP, FP, FN globally, then computing the metric \\
 \choice weighting each class by its frequency \\
 \choice using only the majority class \\
\end{oneparchoices}

\answer{B. Micro-averaging aggregates all TP, FP, FN across classes before computing the metric. Option A describes macro-averaging. \textit{(L06 p42, p44)}}
\vspace{\stretch{1}}

% 출처: L07 (Receptive field) — p85-86
\part[4] What is the receptive field size of three stacked $3 \times 3$ convolution layers? (stride = 1 for all layers)

\begin{oneparchoices}
 \choice $5 \times 5$
 \choice $7 \times 7$
 \choice $9 \times 9$
 \choice $11 \times 11$
\end{oneparchoices}

\answer{B. Each additional $3 \times 3$ layer adds 2 to the receptive field: $3 \rightarrow 5 \rightarrow 7$. (Formula: $(n-1) \times k + 1$ where $n$ = layers, but simpler: $3 + 2 + 2 = 7$). \textit{(L07 p85--86)}}
\vspace{\stretch{1}}

% 출처: L07 (VGGNet) — p85, p87-88
\part[4] What is the main advantage of using three stacked $3 \times 3$ conv layers instead of a single $7 \times 7$ conv layer (with the same number of channels $C$)?

\begin{oneparchoices}
 \choice Larger receptive field \\
 \choice Fewer parameters and more non-linearity \\
 \choice No padding is required \\
 \choice Faster inference speed \\
\end{oneparchoices}

\answer{B. Three $3 \times 3$ layers use $3 \times (9C^2) = 27C^2$ parameters vs.\ $49C^2$ for one $7 \times 7$ layer, and introduce 3 ReLU activations instead of 1. \textit{(L07 p85, p87--88)}}
