% 출처: L02 (MSE problems) — p3
\part[3] When using Mean Squared Error (MSE) loss with a sigmoid output for classification, what is the main problem?

\begin{oneparchoices}
 \choice The loss function is not differentiable. \\
 \choice The loss and gradient magnitudes are upper-bounded, making training slow. \\
 \choice MSE cannot handle multi-class problems. \\
 \choice The loss becomes negative for incorrect predictions. \\
\end{oneparchoices}

\answer{B. Since sigmoid outputs are in $(0,1)$ and targets are in $\{0,1\}$, MSE loss $< 1$ and $|\partial\mathcal{L}/\partial \hat{y}| < 2$, resulting in slow training. \textit{(L02 p3)} \textcolor{red}{[Medium]}}
\vspace{\stretch{1}}

% 출처: L02 (Cross Entropy calculation) — p8-9
\part[4] A classifier outputs the softmax probability $[0.1,\; 0.2,\; 0.3,\; 0.4]$ for a 4-class classification problem. If the ground-truth class is the 3rd class, what is the cross-entropy loss?

\begin{oneparchoices}
 \choice $-\log(0.1)$
 \choice $-\log(0.3)$
 \choice $-\log(0.4)$
 \choice $-\sum_{c=1}^{4} \log(p_c)$
\end{oneparchoices}

\answer{B. With a one-hot vector as the ground truth, the cross-entropy loss $= -\log(\hat{p}_{y_i})$. Since class 3 has $\hat{p}_3 = 0.3$, the loss is $-\log(0.3)$. \textit{(L02 p8--9)} \textcolor{red}{[Medium]}}
\vspace{\stretch{1}}

% 출처: L03 (Softmax + CE gradient) — p72-73
\part[4] Given the softmax output $\mathbf{p} = [0.2,\; 0.5,\; 0.3]$ and the ground-truth class is the 2nd class, what is the gradient of the cross-entropy loss with respect to the logit vector $\mathbf{s}$?

\begin{oneparchoices}
 \choice $[0.2,\; -0.5,\; 0.3]$
 \choice $[0.2,\; 0.5,\; -0.7]$
 \choice $[-0.8,\; 0.5,\; 0.3]$
 \choice $[0.2,\; 0.5,\; 0.3]$
\end{oneparchoices}

\answer{A. $\partial \mathcal{L} / \partial \mathbf{s} = \mathbf{p} - \mathbf{y} = [0.2,\; 0.5,\; 0.3] - [0,\; 1,\; 0] = [0.2,\; -0.5,\; 0.3]$. \textit{(L03 p72--73)} \textcolor{red}{[Hard]}}
\vspace{\stretch{1}}

% 출처: L03 (Max gate) — p28
\part[3] In a computational graph, a max gate routes the upstream gradient to \rule{1cm}{0.15mm}.

\begin{oneparchoices}
 \choice both inputs equally
 \choice the larger input only
 \choice the smaller input only
 \choice neither input
\end{oneparchoices}

\answer{B. The max gate passes the gradient only to the input that was selected (the larger one) and assigns zero gradient to the other input. \textit{(L03 p28)} \textcolor{red}{[Easy]}}
\vspace{\stretch{1}}

% 출처: L04 (Optimizer identification) — p23-24
\part[4] Which optimizer uses an \textbf{exponential moving average of squared gradients} to adapt the learning rate for each parameter?

\begin{oneparchoices}
 \choice SGD with Momentum
 \choice AdaGrad
 \choice RMSProp
 \choice Nesterov Momentum
\end{oneparchoices}

\answer{C. RMSProp uses $v_t = \rho \, v_{t-1} + (1-\rho) \, g_t^2$ (exponential moving average). AdaGrad uses a cumulative sum, not a moving average. Momentum and Nesterov do not adapt per-parameter learning rates. \textit{(L04 p23--24)} \textcolor{red}{[Medium]}}
\vspace{\stretch{1}}

% 출처: L04 (Mini-batch SGD) — p4-5
\part[3] What is the main advantage of using mini-batch SGD over full-batch gradient descent?

\begin{oneparchoices}
 \choice It always converges to the global minimum. \\
 \choice It is computationally faster per update by approximating the gradient. \\
 \choice It computes the exact gradient of the full dataset. \\
 \choice It eliminates the need for a learning rate. \\
\end{oneparchoices}

\answer{B. Mini-batch SGD approximates the gradient using a subset, making each update much faster than computing the gradient over the entire dataset. \textit{(L04 p4--5)} \textcolor{red}{[Easy]}}
\vspace{\stretch{1}}

% 출처: L05 (Dead neuron) — p2, p9
\part[3] Which activation function can suffer from the ``dead neuron'' problem?

\begin{oneparchoices}
 \choice Sigmoid
 \choice Tanh
 \choice ReLU
 \choice Softmax
\end{oneparchoices}

\answer{C. ReLU outputs 0 for all negative inputs and its gradient is 0 in that region, so a neuron can become permanently dead. Sigmoid and Tanh have vanishing but never exactly zero gradients. \textit{(L05 p2, p9)} \textcolor{red}{[Easy]}}
\vspace{\stretch{1}}

% 출처: L06 (F1 score calculation) — p4-5, p12
\part[4] A binary classifier produces the following results: $\text{TP} = 40$, $\text{FP} = 10$, $\text{FN} = 10$, $\text{TN} = 40$. What is the F1 score?

\begin{oneparchoices}
 \choice 0.75
 \choice 0.80
 \choice 0.85
 \choice 0.90
\end{oneparchoices}

\answer{B. Precision $= 40/50 = 0.8$, Recall $= 40/50 = 0.8$, $F_1 = 2 \times 0.8 \times 0.8 / (0.8 + 0.8) = 0.80$. \textit{(L06 p4--5, p12)} \textcolor{red}{[Medium]}}
\vspace{\stretch{1}}

% 출처: L07 (Convolution output size) — p54-55
\part[4] Given an input of size $32 \times 32 \times 3$, what is the output size after a convolution layer with 16 filters of size $5 \times 5$, stride 1, and padding 0?

\begin{oneparchoices}
 \choice $28 \times 28 \times 3$
 \choice $28 \times 28 \times 16$
 \choice $32 \times 32 \times 16$
 \choice $28 \times 28 \times 5$
\end{oneparchoices}

\answer{B. Output spatial size $= (32 - 5)/1 + 1 = 28$. Output depth $=$ number of filters $= 16$. So $28 \times 28 \times 16$. \textit{(L07 p54--55)} \textcolor{red}{[Medium]}}

 \choice 0
 \choice 1
 \choice 2
 \choice 3

% 출처: L04 (Adam bias correction) — p26-28
\part[3] In the Adam optimizer, why is bias correction applied to the moving average estimates?

\begin{oneparchoices}
 \choice To prevent division by zero in the denominator \\
 \choice To ensure the sum of weights in the moving average equals 1 \\
 \choice To gradually increase the learning rate over time \\
 \choice To normalize the gradient magnitude to unit length \\
\end{oneparchoices}

\answer{B. When $t$ is small, the sum of moving average weights $(1-\beta^t)$ is less than 1. Bias correction divides by $(1-\beta^t)$ to compensate. \textit{(L04 p26--28)} \textcolor{red}{[Med-High]}}
\vspace{\stretch{1}}

% 출처: L03 (Linear layer backprop dL/dW) — p66-67
\part[3] For a linear layer $\mathbf{y} = W\mathbf{x}$, the gradient of the loss with respect to the weight matrix $W$ is:

\begin{oneparchoices}
 \choice $W^\top \frac{d\mathcal{L}}{d\mathbf{y}}$
 \choice $\frac{d\mathcal{L}}{d\mathbf{y}} \cdot \mathbf{x}^\top$
 \choice $\mathbf{x} \cdot \left(\frac{d\mathcal{L}}{d\mathbf{y}}\right)^\top$
 \choice $\frac{d\mathcal{L}}{d\mathbf{y}} \cdot W$
\end{oneparchoices}

\answer{B. $\frac{d\mathcal{L}}{dW} = \frac{d\mathcal{L}}{d\mathbf{y}} \cdot \mathbf{x}^\top$. Note: option A is $\frac{d\mathcal{L}}{d\mathbf{x}}$, not $\frac{d\mathcal{L}}{dW}$. \textit{(L03 p66--67)} \textcolor{red}{[Med-High]}}
\vspace{\stretch{1}}

% 출처: L03 (Add gate) — p28
\part[3] An add gate in a computational graph works as a gradient \rule{1cm}{0.15mm}.

\begin{oneparchoices}
 \choice distributor
 \choice switcher
 \choice amplifier
 \choice None of the above
\end{oneparchoices}

\part[4] Which of the following is \textbf{NOT} true about the Adam optimizer?

\begin{oneparchoices}

 \choice It combines AdaGrad and Momentum. \\
 \choice It applies bias correction to the moment estimates. \\

\part[3] During test time, a neural network with inverted dropout \rule{1cm}{0.15mm}.

 \choice uses all neurons without any modification \\
 \choice scales the weights by the dropout rate \\
 \choice removes the dropout layers entirely from the network \\

% 출처: L02 (KL Divergence vs Cross Entropy) — p10
\part[3] Minimizing the cross-entropy loss $H(P, Q)$ is equivalent to minimizing the KL divergence $D_{KL}(P \| Q)$ because:

\begin{oneparchoices}
 \choice They are the same function. \\
 \choice They differ only by the entropy $H(P)$, which is constant w.r.t.\ model parameters. \\
 \choice KL divergence is always zero for one-hot distributions. \\
 \choice Cross-entropy ignores the ground-truth distribution $P$. \\
\end{oneparchoices}

\answer{B. $H(P,Q) = D_{KL}(P \| Q) + H(P)$. Since $P$ is the fixed ground-truth distribution, $H(P)$ is a constant that does not affect optimization. \textit{(L02 p10)} \textcolor{red}{[Medium]}}
