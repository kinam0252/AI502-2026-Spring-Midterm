\question[70] For each of the following statements, select the most proper answer of each question.

\begin{parts}

% 출처: L01 (MLP, XOR)
\part[5] What is the minimum number of layers (counting hidden and output layers only) required for a neural network to solve the XOR problem?

\begin{oneparchoices}
 \choice 1
 \choice 2
 \choice 3
 \choice 4
\end{oneparchoices}

\answer{B. XOR requires at least 1 hidden layer + 1 output layer = 2 layers.}
\vspace{\stretch{1}}

% 출처: L02 (Cross Entropy calculation)
\part[7] A classifier outputs the softmax probability $[0.1,\; 0.2,\; 0.3,\; 0.4]$ for a 4-class classification problem. If the ground-truth class is class 3, what is the cross-entropy loss?

\begin{oneparchoices}
 \choice $-\log(0.1)$
 \choice $-\log(0.3)$
 \choice $-\log(0.4)$
 \choice $-\sum_{c=1}^{4} \log(p_c)$
\end{oneparchoices}

\answer{B. With one-hot ground truth, CE loss $= -\log(\hat{p}_{y_i})$. Since class 3 has $\hat{p}_3 = 0.3$, the loss is $-\log(0.3)$.}
\vspace{\stretch{1}}

% 출처: L03 (Softmax + CE gradient, 값 변형)
\part[8] Given the softmax output $\mathbf{p} = [0.2,\; 0.5,\; 0.3]$ and the ground-truth class is class 2, what is the gradient of the cross-entropy loss with respect to the logit vector $\mathbf{s}$?

\begin{oneparchoices}
 \choice $[0.2,\; -0.5,\; 0.3]$
 \choice $[0.2,\; 0.5,\; -0.7]$
 \choice $[-0.8,\; 0.5,\; 0.3]$
 \choice $[0.2,\; 0.5,\; 0.3]$
\end{oneparchoices}

\answer{A. $\partial \mathcal{L} / \partial \mathbf{s} = \mathbf{p} - \mathbf{y} = [0.2,\; 0.5,\; 0.3] - [0,\; 1,\; 0] = [0.2,\; -0.5,\; 0.3]$.}
\vspace{\stretch{1}}

% 출처: L03 (Max gate)
\part[5] In a computational graph, a max gate routes the upstream gradient to \rule{1cm}{0.15mm}.

\begin{oneparchoices}
 \choice both inputs equally
 \choice the larger input only
 \choice the smaller input only
 \choice neither input
\end{oneparchoices}

\answer{B. The max gate passes the gradient only to the input that was selected (the larger one) and assigns zero gradient to the other input.}
\vspace{\stretch{1}}

% 출처: L04 (Optimizer identification)
\part[7] Which optimizer uses an \textbf{exponential moving average of squared gradients} to adapt the learning rate for each parameter?

\begin{oneparchoices}
 \choice SGD with Momentum
 \choice AdaGrad
 \choice RMSProp
 \choice Nesterov Momentum
\end{oneparchoices}

\answer{C. RMSProp uses $v_t = \rho \, v_{t-1} + (1-\rho) \, g_t^2$ (exponential moving average). AdaGrad uses a cumulative sum, not a moving average. Momentum and Nesterov do not adapt per-parameter learning rates.}
\vspace{\stretch{1}}

% 출처: L04 (Mini-batch SGD)
\part[5] What is the main advantage of using mini-batch SGD over full-batch gradient descent?

\begin{oneparchoices}
 \choice It always converges to the global minimum. \\
 \choice It requires less memory and allows more frequent parameter updates. \\
 \choice It computes the exact gradient of the full dataset. \\
 \choice It eliminates the need for a learning rate. \\
\end{oneparchoices}

\answer{B. Mini-batch SGD processes a subset of data at a time, requiring less memory and enabling more frequent updates per epoch.}
\vspace{\stretch{1}}

% 출처: L05 (Dead neuron)
\part[7] Which activation function can suffer from the ``dead neuron'' problem?

\begin{oneparchoices}
 \choice Sigmoid
 \choice Tanh
 \choice ReLU
 \choice Softmax
\end{oneparchoices}

\answer{C. ReLU outputs 0 for all negative inputs and its gradient is 0 in that region, so a neuron can become permanently dead. Sigmoid and Tanh have vanishing but never exactly zero gradients.}
\vspace{\stretch{1}}

% 출처: L06 (F1 score calculation)
\part[8] A binary classifier produces the following results: $\text{TP} = 40$, $\text{FP} = 10$, $\text{FN} = 10$, $\text{TN} = 40$. What is the F1 score?

\begin{oneparchoices}
 \choice 0.75
 \choice 0.80
 \choice 0.85
 \choice 0.90
\end{oneparchoices}

\answer{B. Precision $= 40/50 = 0.8$, Recall $= 40/50 = 0.8$, $F_1 = 2 \times 0.8 \times 0.8 / (0.8 + 0.8) = 0.80$.}
\vspace{\stretch{1}}

% 출처: L07 (Convolution output size)
\part[8] Given an input of size $32 \times 32 \times 3$, what is the output size after a convolution layer with 16 filters of size $5 \times 5$, stride 1, and padding 0?

\begin{oneparchoices}
 \choice $28 \times 28 \times 3$
 \choice $28 \times 28 \times 16$
 \choice $32 \times 32 \times 16$
 \choice $28 \times 28 \times 5$
\end{oneparchoices}

\answer{B. Output spatial size $= (32 - 5)/1 + 1 = 28$. Output depth $=$ number of filters $= 16$. So $28 \times 28 \times 16$.}
\vspace{\stretch{1}}

% 출처: L07 (Same padding)
\part[10] What padding size is required for a $3 \times 3$ filter to produce an output with the same spatial dimensions as the input? (stride = 1)

\begin{oneparchoices}
 \choice 0
 \choice 1
 \choice 2
 \choice 3
\end{oneparchoices}

\answer{B. Same padding: $p = (k-1)/2 = (3-1)/2 = 1$.}
\vspace{\stretch{1}}

\end{parts}
